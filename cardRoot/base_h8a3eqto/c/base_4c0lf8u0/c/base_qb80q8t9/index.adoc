= Score card widget

== Built-in score card macro, for comparison

{{#scoreCard}}
  "title": "Open findings",
  "value": 42,
  "legend": "across all projects"
{{/scoreCard}}

== scoreCardWidget report

{{#report}}
  "name": "base/reports/scoreCardWidget",
  "title": "Open findings",
  "value": 42,
  "legend": "across all projects"
{{/report}}

== With a unit

{{#report}}
  "name": "base/reports/scoreCardWidget",
  "title": "Mean time to remediate",
  "value": 12.5,
  "unit": "days",
  "legend": "over the last quarter"
{{/report}}

== Percent unit, no space

  "value": 93,
  "unit": "%",
  "legend": "of all controls"
{{/report}}
